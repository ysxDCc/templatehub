Stream Brainwaves in 4 Lines of Code

Our zero-overhead binary IPC client streams decoded spiking probabilities directly into PyTorch, TensorRT, or ROS2 robotics pipelines.

import aether_neuro as an

# Initialize 10-20 telemetry client via PCIe/USB4 optical link
session = an.StreamSession(channels=1024, sample_rate_hz=16384)
session.calibrate_impedance(target_kohms=2.0)

# Real-time intent classification at 0.42ms latency
for packet in session.stream_cortical_packets():
    intent = an.decode_kinematics(packet.motor_cortex_spikes)
    print(f"Decoded Velocity Vector: {intent.dx:.3f}, {intent.dy:.3f} | Confidence: {intent.confidence:.2%}")
